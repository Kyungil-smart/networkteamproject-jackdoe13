# LifeCycleManager.cs

```csharp
using UnityEngine;
using UnityEngine.InputSystem;
public class LifeCycleManager : MonoBehaviour
{
    [Header("제어")]
    [SerializeField] private GameObject targetObject;

    [Header("Player")]
    [SerializeField] private PlayerInput playerInput;

    [Header("점프")]
    [SerializeField] private float jumpForce = 5f;

    private InputAction toggleAction;
    private InputAction destroyAction;
    private InputAction jumpAction;

    private void Awake()
    {
        Debug.Log("Awake(LifeCycleManager) 호출");

        if (playerInput == null)
        {
            playerInput = GetComponent<PlayerInput>();
        }

        if (playerInput == null)
        {
            Debug.LogError("PlayerInput 없습니다.");
            return;
        }

        jumpAction = playerInput.actions.FindAction("Jump");
        toggleAction = playerInput.actions.FindAction("ToggleActive");
        destroyAction = playerInput.actions.FindAction("DestroyObject");
    }

    private void OnEnable()
    {
        Debug.Log("OnEnable(LifeCycleManager) 호출");

        if (jumpAction != null)
        {
            jumpAction.performed += OnJump;
        }

        if (toggleAction != null)
        {
            toggleAction.performed += OnToggle;
        }

        if (destroyAction != null)
        {
            destroyAction.performed += OnDestroyTarget;
        }
    }

    private void OnDisable()
    {
        Debug.Log("OnDisable(LifeCycleManager) 호출");

        if (jumpAction != null)
        {
            jumpAction.performed -= OnJump;
        }

        if (toggleAction != null)
        {
            toggleAction.performed -= OnToggle;
        }

        if (destroyAction != null)
        {
            destroyAction.performed -= OnDestroyTarget;
        }
    }

    private void OnJump(InputAction.CallbackContext context)
    {
        Debug.Log("Space 입력 - Jump");

        if (targetObject == null) return;

        LifeCyclePhysicsTest physicsTest = targetObject.GetComponent<LifeCyclePhysicsTest>();

        if (physicsTest != null)
        {
            physicsTest.Jump(jumpForce);
        }
    }

    private void OnToggle(InputAction.CallbackContext context)
    {
        if (targetObject == null) return;

        bool nextState = !targetObject.activeSelf;
        targetObject.SetActive(nextState);

        Debug.Log("1번 입력 - Target Active 상태: " + nextState);
    }

    private void OnDestroyTarget(InputAction.CallbackContext context)
    {
        if (targetObject == null) return;

        Debug.Log("2번 입력 - Target 삭제");

        Destroy(targetObject);
        targetObject = null;
    }
}
```

```csharp
using Unity.VisualScripting;
using UnityEngine;
using UnityEngine.InputSystem;

public class LifeCyclePhysicsTest : MonoBehaviour
{
    [Header("물리")]
    [SerializeField] private float moveForce = 5f;

    private Rigidbody rb;

    private void Awake()
    {
        Debug.Log("Awake(LifeCyclePhysicsTest): Rigidbody 초기화");

        rb = GetComponent<Rigidbody>();

        if (rb == null)
        {
            Debug.LogError("Rigidbody가 없습니다.");
        }
    }

    private void OnEnable()
    {
        Debug.Log("OnEnable(LifeCyclePhysicsTest) 호출");
    }

    private void Start()
    {
        Debug.Log("Start(LifeCyclePhysicsTest) 호출");
    }

    private void FixedUpdate()
    {
        if (rb != null)
        {
            rb.AddForce(Vector3.forward * moveForce);
        }
    }

    private void Update()
    {
    }

    private void LateUpdate()
    {
    }

    private void OnDisable()
    {
        Debug.Log("OnDisable(LifeCyclePhysicsTest) 호출");
    }

    private void OnDestroy()
    {
        Debug.Log("OnDestroy(LifeCyclePhysicsTest) 오브젝트 삭제");
    }

    public void Jump(float jumpForce)
    {
        Debug.Log("Jump(LifeCyclePhysicsTest) 호출");

        if (rb != null)
        {
            rb.AddForce(Vector3.up * jumpForce, ForceMode.Impulse);
        }
    }
}
```

```csharp
using UnityEngine;

public class LifeCycleCameraFollow : MonoBehaviour
{
    [SerializeField] private Transform target;

    [SerializeField] private Vector3 offset = new Vector3(0f, 4f, -7f);

    [SerializeField] private bool showLateUpdateLog = true;

    private void Awake()
    {
        Debug.Log("Awake 호출, 카메라 초기화");
    }

    private void Start()
    {
        Debug.Log("Start 호출, 추적 대상 확인");

        if(target == null)
        {
            Debug.LogError("카메라에 추적 대상이 설정되지 않았습니다.");
        }
    }

    private void LateUpdate()
    {
        if (target == null) return;

        transform.position = target.position + offset;
        transform.LookAt(target);

        if(showLateUpdateLog)
        {

            Debug.Log("LateUpdate 호출, 카메라로 플레이어 추적");
        }
    }
}
```
